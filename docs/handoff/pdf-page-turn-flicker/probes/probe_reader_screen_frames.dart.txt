Future<String> _frameHash(WidgetTester tester) async {
  final boundary = tester.renderObject<RenderRepaintBoundary>(
    find.byType(RepaintBoundary).first,
  );
  final image = await boundary.toImage(pixelRatio: 0.3);
  final w = image.width, hgt = image.height;
  final data = await image.toByteData(format: ui.ImageByteFormat.rawRgba);
  image.dispose();
  final bytes = data!.buffer.asUint8List();
  int hashRows(int from, int to) {
    var h = 0x811c9dc5;
    for (var i = from * w * 4; i < to * w * 4; i++) {
      h = ((h ^ bytes[i]) * 0x01000193) & 0xffffffff;
    }
    return h;
  }
  final top = (hgt * 0.12).round(), bot = (hgt * 0.88).round();
  String f(int v) => (v & 0xffff).toRadixString(16).padLeft(4, '0');
  return 'T:${f(hashRows(0, top))} M:${f(hashRows(top, bot))} B:${f(hashRows(bot, hgt))}';
}

            await Future<void>.delayed(const Duration(milliseconds: 25));
